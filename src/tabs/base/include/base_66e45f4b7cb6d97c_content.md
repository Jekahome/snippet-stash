fn min<'a>(x:&'a i32, y: &'a i32) -> i32{
    if x < y{
        x
    } else{
        y
    }
}
fn main(){
    let p = 42; //--------------------------------------------------+ 
    { //                                                            | p's lifetime
        let q = 10; //-------------------------------+              |
        let r = min(&p, &q); //-------+              | q's lifetime |
        println!("Min is {}", r); //  | r's lifetime |              |
    } //------------------------------+--------------+              |
}//-----------------------------------------------------------------+
