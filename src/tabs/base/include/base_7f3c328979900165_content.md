use std::convert::From;
struct A{
   field_a:i32
}

#[derive(Debug)]
struct B{
   field_b:i32
}

impl From<A> for B {
    fn from(a: A) -> Self {
        B{field_b:a.field_a}
    }
}

fn main(){
    // `impl From for A`
    let a:A = A{field_a:8};
    let b:B = From::from(a);
    println!("{:?}",b);// B { field_b: 8 }

    // Into автореализация `impl Into<A> for B`
    let a:A = A{field_a:8};
    let b:B = a.into();
    println!("{:?}",b);// B { field_b: 8 }
}
