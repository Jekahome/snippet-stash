

Почему правило заимствования позволяет иметь только одну мутабельную ссылку или множество немутабельных
Из-за возможной некорректности программы в процессе выполнения и компилятор лучше оптимизирует код зная что мутабельная ссылка одна
```rust
fn main(){
    let mut a = 2;
    let mut b = 0;
    foo(&mut a,&mut b);
    assert_eq!(10,b);
   
// Но если бы можно было иметь больше одной мутабельной ссылки
// то мы бы могли передать мутабельную ссылку на одни и те же данные и в итоге меняли их но с разных ссылок
    foo(&mut a,&mut a);
    assert_eq!(100,b);
}
  
fn foo(a:&mut i32,b:&mut i32){
    if *a > 1 {
        *b = 10;
    }
    if *a > 5 {
        *b = 100;
    }
}
```
