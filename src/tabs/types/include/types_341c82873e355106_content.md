trait Say{
    fn say(&self);
}
struct Dog;
struct Cat;
 
impl Say for Dog{
    fn say(&self){println!("Dog gav");}
} 
impl Say for Cat{
    fn say(&self){println!("Cat may");}
} 
 fn get()->Box<dyn Say> {
      if true { Box::new(Cat) }else{ Box::new(Dog)}
}
fn main() {
 let cat:Cat = Cat;
 let cat:&dyn Say = &cat;

 let dog:Dog = Dog;
 let dog:&dyn Say = &dog;
   
 let animals:Vec<&dyn Say> = vec![cat,dog];  
 
 for animal in animals.iter(){
     animal.say();
 } 
}
