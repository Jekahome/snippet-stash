use std::process::Command;
fn main(){
    let mut cmd = Command::new("python");
    cmd.arg("src/bin/decode.py");

    // Execute the command
    match cmd.output() {
        Ok(o) => {
            let res:String = unsafe{
                String::from_utf8_unchecked(o.stdout)

            };
            println!("Output: {}",res);
        },
        Err(e) => {
            eprintln!("There was an error! {}",e);
            process::exit(1);
        }
    }
}
